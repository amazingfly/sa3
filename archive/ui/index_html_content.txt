<script>
const form = document.getElementById('generateForm');
const jobsEl = document.getElementById('jobs');
const statusEl = document.getElementById('serverStatus');
const generateBtn = document.getElementById('generateBtn');
const autoRefreshEl = document.getElementById('autoRefresh');
const modelNameEl = document.getElementById('modelName');
const durationEl = document.getElementById('duration');
const continuationModeEl = document.getElementById('continuationMode');
const rawConfigEl = document.getElementById('rawConfig');
const loadDefaultBtn = document.getElementById('loadDefaultBtn');

let knownJobs = [];
let pollTimer = null;
let currentMode = 'standardInputs';

// Tab logic
document.querySelectorAll('.tab').forEach(tab => {
  tab.addEventListener('click', () => {
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    tab.classList.add('active');
    currentMode = tab.dataset.target;
    document.getElementById('standardInputs').style.display = currentMode === 'standardInputs' ? 'block' : 'none';
    document.getElementById('configInput').style.display = currentMode === 'configInput' ? 'block' : 'none';
  });
});

loadDefaultBtn.addEventListener('click', async () => {
  try {
    const data = await api('/api/defaults');
    rawConfigEl.value = JSON.stringify(data, null, 2);
  } catch (err) {
    alert("Failed to load defaults: " + err.message);
  }
});

function esc(s) {
  return String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
}

async function api(path, options) {
  const res = await fetch(path, options);
  const data = await res.json();
  if (!res.ok) throw new Error(data.error || res.statusText);
  return data;
}

function ensureJobElement(job) {
  let el = jobsEl.querySelector(`[data-job-id="${job.id}"]`);
  if (el) return el;
  el = document.createElement('article');
  el.className = 'job';
  el.dataset.jobId = job.id;
  el.innerHTML = `
    <div class="job-head">
      <div>
        <strong class="job-title"></strong>
        <div class="muted job-created"></div>
      </div>
      <span class="badge job-status"></span>
    </div>
    <div class="job-audio"></div>
    <div class="muted job-error"></div>
    <pre class="job-log"></pre>
  `;
  jobsEl.prepend(el);
  return el;
}

function updateJobElement(job) {
  const el = ensureJobElement(job);
  const badgeClass = job.status === 'done' ? 'done' : job.status === 'failed' ? 'failed' : 'running';
  el.querySelector('.job-title').textContent = job.track_id || job.id;
  el.querySelector('.job-created').textContent = job.created_at || '';
  const badge = el.querySelector('.job-status');
  badge.className = `badge job-status ${badgeClass}`;
  badge.textContent = job.status;
  el.querySelector('.job-error').textContent = job.error || '';
  const log = el.querySelector('.job-log');
  if (log.textContent !== (job.log_tail || '')) {
    const wasAtBottom = Math.abs(log.scrollHeight - log.clientHeight - log.scrollTop) < 8;
    log.textContent = job.log_tail || '';
    if (wasAtBottom) log.scrollTop = log.scrollHeight;
  }
  const audioBox = el.querySelector('.job-audio');
  if (job.status === 'done' && job.audio_url && !audioBox.querySelector('audio')) {
    const audio = document.createElement('audio');
    audio.controls = true;
    audio.src = job.audio_url;
    const link = document.createElement('a');
    link.href = job.audio_url;
    link.download = '';
    link.textContent = 'Download OGG';
    audioBox.append(audio, link);
    if (job.flac_path) {
      const flac = document.createElement('span');
      flac.className = 'muted';
      flac.textContent = ` FLAC: ${job.flac_path}`;
      audioBox.append(flac);
    }
  }
}

function renderJobs(jobs) {
  jobs.slice().reverse().forEach(updateJobElement);
  generateBtn.disabled = jobs.some(j => j.status === 'queued' || j.status === 'running');
}

function startPolling() {
  if (pollTimer === null && autoRefreshEl.checked) {
    pollTimer = setInterval(refresh, 5000);
  }
}

function stopPolling() {
  if (pollTimer !== null) {
    clearInterval(pollTimer);
    pollTimer = null;
  }
}

async function refresh() {
  try {
    const data = await api('/api/jobs');
    knownJobs = data.jobs;
    renderJobs(knownJobs);
    statusEl.textContent = data.running ? 'Generating' : 'Ready';
    if (data.running && autoRefreshEl.checked) {
      startPolling();
    } else if (!data.running) {
      stopPolling();
      if (knownJobs.length && knownJobs[0].status === 'done') {
        statusEl.textContent = 'Finished';
      }
    }
  } catch (err) {
    statusEl.textContent = err.message;
  }
}

form.addEventListener('submit', async (event) => {
  event.preventDefault();
  console.log("Submitting form, mode:", currentMode);
  let payload;
  if (currentMode === 'configInput') {
    const val = rawConfigEl.value.trim();
    if (!val) {
      alert("Please paste a JSON config first.");
      return;
    }
    try {
      payload = { raw_config: JSON.parse(val) };
    } catch (e) {
      alert("Invalid JSON config: " + e.message);
      return;
    }
  } else {
    payload = {
      prompt: document.getElementById('prompt').value,
      model: modelNameEl.value,
      continuation: continuationModeEl.checked,
      track_id: document.getElementById('trackId').value,
      duration_seconds: Number(durationEl.value),
      steps: Number(document.getElementById('steps').value),
      cfg_scale: Number(document.getElementById('cfgScale').value),
      init_noise_level: Number(document.getElementById('initNoise').value),
      seed_start: document.getElementById('seed').value ? Number(document.getElementById('seed').value) : null
    };
  }
  
  generateBtn.disabled = true;
  statusEl.textContent = 'Starting...';
  statusEl.style.color = 'var(--warn)';
  
  try {
    console.log("Sending payload:", payload);
    const res = await api('/api/generate', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify(payload)
    });
    console.log("Generate response:", res);
    statusEl.textContent = 'Generating...';
    await refresh();
    startPolling();
  } catch (err) {
    console.error("Generate error:", err);
    statusEl.textContent = 'Error: ' + err.message;
    statusEl.style.color = 'var(--bad)';
    alert("Generation failed: " + err.message);
    generateBtn.disabled = false;
  }
});

document.getElementById('refreshBtn').addEventListener('click', refresh);
modelNameEl.addEventListener('change', () => {
  if (modelNameEl.value === 'medium') {
    continuationModeEl.checked = false;
    durationEl.value = '30';
    durationEl.max = '380';
  } else {
    continuationModeEl.checked = true;
    durationEl.value = '120';
    durationEl.max = '120';
  }
});
autoRefreshEl.addEventListener('change', () => {
  if (autoRefreshEl.checked && knownJobs.some(j => j.status === 'queued' || j.status === 'running')) {
    startPolling();
  } else {
    stopPolling();
  }
});
refresh();
</script>
</body>
</html>
